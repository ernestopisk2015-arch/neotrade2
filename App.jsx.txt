import { useState, useRef } from 'react'

const SYSTEM_PROMPT = `Eres Luna, Trade Coach y analista tecnica experta en futuros de criptomonedas y acciones tokenizadas. Trabajas para NEOTRADE, la plataforma de educacion financiera con IA mas avanzada de Latinoamerica.

const A = [{n:'TSLA',p:'422.51',c:'+4.47%',u:true},{n:'NVDA',p:'223.30',c:'+0.60%',u:true},{n:'BTC',p:'106,800',c:'+1.20%',u:true},{n:'ETH',p:'2,540',c:'+2.10%',u:true},{n:'SPY',p:'587.40',c:'+0.85%',u:true}]

export default function App(){
  const [sel,setSel]=useState('TSLA')
  const [tf,setTf]=useState('4h')
  const [inp,setInp]=useState('')
  const [loading,setLoading]=useState(false)
  const [images,setImages]=useState([])
  const [imagesB64,setImagesB64]=useState([])
  const fileRef=useRef()
  const [msgs,setMsgs]=useState([{f:'l',t:'Hola! Soy Luna tu Trade Coach de NEOTRADE. Puedes preguntarme sobre cualquier activo o subir varias imagenes de charts para que las analice en conjunto.'}])

  const handleImages=(e)=>{
    const files=Array.from(e.target.files)
    if(!files.length)return
    const previews=files.map(f=>URL.createObjectURL(f))
    setImages(prev=>[...prev,...previews])
    files.forEach(file=>{
      const reader=new FileReader()
      reader.onload=(ev)=>{
        const b64=ev.target.result.split(',')[1]
        setImagesB64(prev=>[...prev,b64])
      }
      reader.readAsDataURL(file)
    })
  }

  const removeImage=(i)=>{
    setImages(prev=>prev.filter((_,idx)=>idx!==i))
    setImagesB64(prev=>prev.filter((_,idx)=>idx!==i))
  }

  const send=async()=>{
    if((!inp.trim()&&!imagesB64.length)||loading)return
    const userMsg=inp||'Analiza estos charts'
    const userImages=[...images]
    const b64s=[...imagesB64]
    setMsgs(m=>[...m,{f:'u',t:userMsg,imgs:userImages}])
    setInp('')
    setImages([])
    setImagesB64([])
    setLoading(true)
    try{
      const history=msgs.slice(-6).map(m=>({role:m.f==='l'?'assistant':'user',content:m.t}))
      let userContent
      if(b64s.length>0){
        userContent=[
          ...b64s.map(b64=>({type:'image_url',image_url:{url:`data:image/jpeg;base64,${b64}`}})),
          {type:'text',text:userMsg||'Analiza estos charts de trading en conjunto y dame un analisis multi-timeframe completo con entrada, stop loss y target.'}
        ]
      } else {
        userContent=userMsg
      }
      const res=await fetch('http://104.218.49.83:3005/api/luna',{
        method:'POST',
        headers:{'Content-Type':'application/json'},
        body:JSON.stringify({
          model:'meta-llama/llama-4-scout-17b-16e-instruct',
          messages:[{role:'system',content:SYSTEM_PROMPT},...history,{role:'user',content:userContent}],
          max_tokens:800,
          temperature:0.7
        })
      })
      const data=await res.json()
      const reply=data.choices?.[0]?.message?.content||'Lo siento, hubo un error. Intenta de nuevo.'
      setMsgs(m=>[...m,{f:'l',t:reply}])
    }catch(e){
      setMsgs(m=>[...m,{f:'l',t:'Error de conexion. Intenta de nuevo.'}])
    }
    setLoading(false)
  }

  return(
    <div style={{fontFamily:'sans-serif',minHeight:'100vh',background:'#0d0d0d',color:'#e0e0e0'}}>
      <div style={{display:'flex',alignItems:'center',justifyContent:'space-between',padding:'14px 24px',borderBottom:'1px solid #222'}}>
        <div style={{fontSize:20,fontWeight:600,letterSpacing:2}}>NEO<span style={{color:'#c9a84c'}}>TRADE</span></div>
        <div style={{display:'flex',gap:8}}>
          {['Dashboard','Senales','Luna','Suscripcion'].map(n=><button key={n} style={{fontSize:12,padding:'5px 14px',borderRadius:6,border:'1px solid #333',background:'transparent',color:'#aaa',cursor:'pointer'}}>{n}</button>)}
        </div>
      </div>
      <div style={{display:'grid',gridTemplateColumns:'repeat(5,1fr)',gap:10,padding:'16px 24px'}}>
        {A.map(a=><div key={a.n} onClick={()=>setSel(a.n)} style={{background:sel===a.n?'#1a1a1a':'#111',border:sel===a.n?'1px solid #c9a84c':'1px solid #222',borderRadius:8,padding:12,cursor:'pointer'}}><div style={{fontSize:11,color:'#888'}}>{a.n}</div><div style={{fontSize:16,fontWeight:600}}>${a.p}</div><div style={{fontSize:12,color:a.u?'#1D9E75':'#D85A30'}}>{a.c}</div></div>)}
      </div>
      <div style={{display:'grid',gridTemplateColumns:'1fr 340px',gap:12,padding:'0 24px 24px'}}>
        <div style={{background:'#111',border:'1px solid #222',borderRadius:12,padding:16}}>
          <div style={{fontSize:14,fontWeight:600,marginBottom:12}}>Analisis - {sel}</div>
          <div style={{display:'flex',gap:6,marginBottom:12}}>
            {['1m','15m','1h','4h','1D'].map(t=><button key={t} onClick={()=>setTf(t)} style={{fontSize:11,padding:'4px 10px',borderRadius:6,border:'1px solid #333',background:tf===t?'#1a1a1a':'transparent',color:tf===t?'#c9a84c':'#888',cursor:'pointer'}}>{t}</button>)}
          </div>
          <div style={{border:'1px solid #1D9E75',background:'#0a1f17',borderRadius:8,padding:'12px 16px',marginBottom:12}}>
            <div style={{fontSize:11,color:'#1D9E75',fontWeight:600,marginBottom:4}}>Senal activa - largo</div>
            <div style={{fontSize:13,color:'#9FE1CB'}}>Estructura alcista confirmada. MAs alineadas. RSI sano.</div>
          </div>
          <div style={{display:'grid',gridTemplateColumns:'1fr 1fr 1fr',gap:8,marginBottom:12}}>
            <div style={{background:'#0d0d0d',border:'1px solid #222',borderRadius:8,padding:10,textAlign:'center'}}><div style={{fontSize:10,color:'#666'}}>Entrada</div><div style={{fontSize:15,fontWeight:600}}>$414</div></div>
            <div style={{background:'#0d0d0d',border:'1px solid #222',borderRadius:8,padding:10,textAlign:'center'}}><div style={{fontSize:10,color:'#666'}}>Stop Loss</div><div style={{fontSize:15,fontWeight:600,color:'#D85A30'}}>$407</div></div>
            <div style={{background:'#0d0d0d',border:'1px solid #222',borderRadius:8,padding:10,textAlign:'center'}}><div style={{fontSize:10,color:'#666'}}>Target</div><div style={{fontSize:15,fontWeight:600,color:'#1D9E75'}}>$430</div></div>
          </div>
          <div style={{display:'grid',gridTemplateColumns:'repeat(5,1fr)',gap:8}}>
            {[['RSI 6','61.8'],['RSI 12','58.2'],['RSI 24','54.1'],['MA5','412.0'],['MA10','409.4']].map(([l,v])=><div key={l} style={{background:'#0d0d0d',border:'1px solid #222',borderRadius:8,padding:8,textAlign:'center'}}><div style={{fontSize:10,color:'#666'}}>{l}</div><div style={{fontSize:13,fontWeight:600}}>{v}</div></div>)}
          </div>
        </div>
        <div style={{background:'#111',border:'1px solid #222',borderRadius:12,display:'flex',flexDirection:'column',height:'540px'}}>
          <div style={{padding:'12px 16px',borderBottom:'1px solid #222',display:'flex',alignItems:'center',gap:10}}>
            <div style={{width:32,height:32,borderRadius:'50%',background:'#1a1040',display:'flex',alignItems:'center',justifyContent:'center',fontSize:16}}>🌙</div>
            <div>
              <div style={{fontSize:13,fontWeight:600}}>Luna</div>
              <div style={{fontSize:11,color:'#1D9E75'}}>Trade Coach - en linea</div>
            </div>
          </div>
          <div style={{padding:12,flex:1,display:'flex',flexDirection:'column',gap:8,overflowY:'auto'}}>
            {msgs.map((m,i)=>(
              <div key={i} style={{display:'flex',flexDirection:'column',alignItems:m.f==='l'?'flex-start':'flex-end'}}>
                {m.imgs&&m.imgs.length>0&&(
                  <div style={{display:'flex',gap:4,flexWrap:'wrap',marginBottom:4,justifyContent:m.f==='l'?'flex-start':'flex-end'}}>
                    {m.imgs.map((img,idx)=><img key={idx} src={img} style={{maxHeight:60,maxWidth:80,borderRadius:4,objectFit:'cover'}} alt="chart"/>)}
                  </div>
                )}
                <div style={{padding:'8px 12px',borderRadius:8,fontSize:12,lineHeight:1.6,background:m.f==='l'?'#0d0d0d':'#1a1040',border:m.f==='l'?'1px solid #222':'1px solid #3d2e8a',color:m.f==='l'?'#ccc':'#b8a9f0',maxWidth:'92%',whiteSpace:'pre-wrap'}}>{m.t}</div>
              </div>
            ))}
            {loading&&<div style={{padding:'8px 12px',borderRadius:8,fontSize:12,background:'#0d0d0d',border:'1px solid #222',color:'#666'}}>Luna analizando...</div>}
          </div>
          {images.length>0&&(
            <div style={{padding:'8px 12px',borderTop:'1px solid #222',display:'flex',gap:6,flexWrap:'wrap'}}>
              {images.map((img,i)=>(
                <div key={i} style={{position:'relative'}}>
                  <img src={img} style={{height:50,width:60,objectFit:'cover',borderRadius:4}} alt="preview"/>
                  <button onClick={()=>removeImage(i)} style={{position:'absolute',top:-4,right:-4,width:16,height:16,borderRadius:'50%',background:'#D85A30',border:'none',color:'white',fontSize:10,cursor:'pointer',display:'flex',alignItems:'center',justifyContent:'center'}}>x</button>
                </div>
              ))}
            </div>
          )}
          <div style={{padding:'10px 12px',borderTop:'1px solid #222',display:'flex',gap:8,alignItems:'center'}}>
            <input ref={fileRef} type="file" accept="image/*" multiple style={{display:'none'}} onChange={handleImages}/>
            <button onClick={()=>fileRef.current.click()} title="Subir charts" style={{fontSize:18,padding:'4px 8px',borderRadius:6,border:'1px solid #333',background:'transparent',color:'#c9a84c',cursor:'pointer'}}>📷</button>
            <input value={inp} onChange={e=>setInp(e.target.value)} onKeyDown={e=>e.key==='Enter'&&send()} placeholder="Preguntale a Luna o sube charts..." style={{flex:1,fontSize:12,padding:'6px 10px',borderRadius:6,border:'1px solid #333',background:'#0d0d0d',color:'#e0e0e0'}}/>
            <button onClick={send} disabled={loading} style={{fontSize:12,padding:'6px 12px',borderRadius:6,border:'1px solid #c9a84c',background:'transparent',color:'#c9a84c',cursor:'pointer',opacity:loading?0.5:1}}>Enviar</button>
          </div>
        </div>
      </div>
    </div>
  )
}
