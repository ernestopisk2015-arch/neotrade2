  const [tab, setTab] = useState('luna') // 'mercado' | 'luna' | 'admin'
  const [sel, setSel] = useState('BTCUSDT')
  const [tf, setTf] = useState('4h')
  const [inp, setInp] = useState('')
  const [loading, setLoading] = useState(false)
  const [images, setImages] = useState([])
  const [imagesB64, setImagesB64] = useState([])
  const [mode, setMode] = useState('long')
  const [prices, setPrices] = useState({})
  const [assets, setAssets] = useState(DEFAULT_ASSETS)
  const [adminAuth, setAdminAuth] = useState(false)
  const [adminPass, setAdminPass] = useState('')
  const [newSymbol, setNewSymbol] = useState('')
  const [newLabel, setNewLabel] = useState('')
  const [adminMsg, setAdminMsg] = useState('')
  const fileRef = useRef()
  const chatRef = useRef()

  // Cargar config del servidor al iniciar
  useEffect(() => {
    const loadConfig = async () => {
      try {
        const res = await fetch('/api/config')
        const data = await res.json()
        if (data.assets && data.assets.length > 0) {
          setAssets(data.assets)
        }
      } catch(e) {}
    }
    loadConfig()
  }, [])

  // Guardar activos en servidor
  const saveAssetsToServer = async (newAssets) => {
    try {
      await fetch('/api/config', {
        method: 'POST',
        headers: {'Content-Type': 'application/json'},
        body: JSON.stringify({assets: newAssets, signals: [], news: []})
      })
    } catch(e) {}
  }

  useEffect(() => {
    const fetchPrices = async () => {
      try {
        const results = await Promise.all(
          assets.map(a =>
            fetch(`https://api.bitget.com/api/v2/spot/market/tickers?symbol=${a.n}`)
              .then(r => r.json()).catch(() => null)
          )
        )
        const np = {}
        results.forEach((data, i) => {
          if(data?.data?.[0]){
            const d = data.data[0]
            const last = parseFloat(d.lastPr)
            const open = parseFloat(d.open24h)
            const chg = open > 0 ? ((last - open) / open * 100) : 0
            np[assets[i].n] = {
              price: last >= 1 ? last.toLocaleString('en', {minimumFractionDigits:2, maximumFractionDigits:2}) : last.toFixed(5),
              change: (chg >= 0 ? '+' : '') + chg.toFixed(2) + '%',
              up: chg >= 0
            }
          }
        })
        setPrices(np)
      } catch(e){}
    }
    fetchPrices()
    const iv = setInterval(fetchPrices, 10000)
    return () => clearInterval(iv)
  }, [assets])

  useEffect(() => {
    if(chatRef.current) chatRef.current.scrollTop = chatRef.current.scrollHeight
  }, [msgs, loading])

  const handleImages = (e) => {
    const files = Array.from(e.target.files)
    if(!files.length) return
    setImages(prev => [...prev, ...files.map(f => URL.createObjectURL(f))])
    files.forEach(file => {
      const r = new FileReader()
      r.onload = (ev) => setImagesB64(prev => [...prev, ev.target.result.split(',')[1]])
      r.readAsDataURL(file)
    })
  }

  const removeImage = (i) => {
    setImages(prev => prev.filter((_,idx) => idx !== i))
    setImagesB64(prev => prev.filter((_,idx) => idx !== i))
  }

  const send = async () => {
    if((!inp.trim() && !imagesB64.length) || loading) return
    const modeLabel = mode === 'short' ? '[MODO SHORT]' : '[MODO LARGO]'
    const userMsg = (inp || 'Analiza estos charts') + ` ${modeLabel}`
    const userImgs = [...images]
    const b64s = [...imagesB64]
    setMsgs(m => [...m, {f:'u', t:userMsg, imgs:userImgs}])
    setInp(''); setImages([]); setImagesB64([])
    setLoading(true)
    try {
      const history = msgs.slice(-6).map(m => ({role:m.f==='l'?'assistant':'user', content:m.t}))
      let userContent
      if(b64s.length > 0){
        const mi = mode === 'short'
          ? 'MODO SHORT: Verifica las 3 condiciones (MA5 bajo MA20 en 1D, RSI(6) bajo 50 en 1D, precio bajo MA10 en 1D). Si no se cumplen las 3 BLOQUEA y explica cuales faltan con numeros exactos.'
          : 'MODO LARGO: Aplica la REGLA CRITICA ANTI-ERROR. Si precio bajo MAs en 1D di claramente que NO es momento de largo con los numeros exactos.'
        userContent = [
          ...b64s.map(b64 => ({type:'image_url', image_url:{url:`data:image/jpeg;base64,${b64}`}})),
          {type:'text', text:(inp||'Analiza estos charts en conjunto, analisis multi-timeframe completo.') + '\n\n' + mi}
        ]
      } else {
        userContent = userMsg + (mode==='short' ? ' Verifica las 3 condiciones obligatorias para SHORT.' : '')
      }
      const res = await fetch('http://104.218.49.83:3005/api/luna', {
        method:'POST', headers:{'Content-Type':'application/json'},
        body: JSON.stringify({
          model:'meta-llama/llama-4-scout-17b-16e-instruct',
          messages:[{role:'system', content:SYSTEM_PROMPT}, ...history, {role:'user', content:userContent}],
          max_tokens:800, temperature:0.3
        })
      })
      const data = await res.json()
      setMsgs(m => [...m, {f:'l', t:data.choices?.[0]?.message?.content||'Error. Intenta de nuevo.'}])
    } catch(e) {
      setMsgs(m => [...m, {f:'l', t:'Error de conexion.'}])
    }
    setLoading(false)
  }

 

  const addAsset = async () => {
    if(!newSymbol||!newLabel){setAdminMsg('Completa ambos campos');return}
    const sym = newSymbol.toUpperCase().trim()
    if(assets.find(a=>a.n===sym)){setAdminMsg('Ya existe');return}
    const na = [...assets, {n:sym, label:newLabel.toUpperCase().trim()}]
    setAssets(na)
    setNewSymbol(''); setNewLabel('')
    await saveAssetsToServer(na)
    setAdminMsg('✅ Guardado en servidor: '+sym)
  }

  const removeAsset = async (sym) => {
    const na = assets.filter(a=>a.n!==sym)
    setAssets(na)
    if(sel===sym) setSel(na[0]?.n||'')
    await saveAssetsToServer(na)
    setAdminMsg('🗑 Eliminado y guardado: '+sym)
  }

  const loginAdmin = () => {
    if(adminPass===ADMIN_PASSWORD){setAdminAuth(true);setAdminMsg('')}
    else setAdminMsg('Contraseña incorrecta')
  }

  const selAsset = assets.find(a=>a.n===sel)
  const selPrice = prices[sel]

  return (
    <div style={S.app}>

      {/* HEADER */}
      <div style={S.header}>
        <div>
          <div style={S.logo}>NEO<span style={S.gold}>TRADE</span></div>
          <div style={{fontSize:9, color:'#333', letterSpacing:2, marginTop:1}}>CRYPTO FUTURES · IA</div>
        </div>
        <div style={{textAlign:'right'}}>
          <div style={{fontSize:11, color:'#1D9E75'}}><span style={S.dot}></span>EN VIVO</div>
          <div style={{fontSize:9, color:'#333', marginTop:2}}>Bitget API · 10s</div>
        </div>
      </div>

      {/* TICKER HORIZONTAL SCROLL */}
      <div style={S.ticker}>
        {assets.map(a => {
          const p = prices[a.n]
          return (
            <div key={a.n} onClick={()=>{setSel(a.n);setTab('mercado')}} style={S.tickerCard(sel===a.n)}>
              <div style={{fontSize:10, color:'#555', marginBottom:2}}>{a.label}</div>
              <div style={{fontSize:13, fontWeight:700, color:'#e0e0e0'}}>{p?'$'+p.price:'...'}</div>
              <div style={{fontSize:10, color:p?.up?'#1D9E75':'#D85A30', marginTop:1}}>{p?p.change:'—'}</div>
            </div>
          )
        })}
      </div>

      {/* CONTENIDO POR TAB */}
      <div style={{flex:1, overflowY:'auto', padding:'12px 14px', paddingBottom:60}}>

        {/* TAB MERCADO */}
        {tab==='mercado' && (
          <div>
            <div style={{...S.card, marginBottom:10}}>
              <div style={{display:'flex', justifyContent:'space-between', alignItems:'center', marginBottom:10}}>
                <div>
                  <div style={{fontSize:16, fontWeight:700}}>{selAsset?.label || sel}</div>
                  <div style={{fontSize:22, fontWeight:700, color:'#e0e0e0'}}>{selPrice?'$'+selPrice.price:'Cargando...'}</div>
                  <div style={{fontSize:13, color:selPrice?.up?'#1D9E75':'#D85A30'}}>{selPrice?.change||'—'} 24h</div>
                </div>
                <div style={{display:'flex', gap:5, flexWrap:'wrap', justifyContent:'flex-end'}}>
                  {['1m','15m','1h','4h','1D'].map(t=>
                    <button key={t} onClick={()=>setTf(t)} style={{fontSize:10, padding:'4px 8px', borderRadius:5, border:`1px solid ${tf===t?'#c9a84c':'#222'}`, background:tf===t?'#1a1a1a':'transparent', color:tf===t?'#c9a84c':'#555', cursor:'pointer'}}>{t}</button>
                  )}
                </div>
              </div>
            </div>

            <div style={S.signalCard}>
              <div style={{fontSize:10, color:'#1D9E75', fontWeight:700, letterSpacing:1, marginBottom:4}}>SEÑAL ACTIVA — LARGO</div>
              <div style={{fontSize:12, color:'#9FE1CB'}}>Estructura alcista confirmada. MAs alineadas. RSI sano.</div>
            </div>

            <div style={S.statGrid}>
              <div style={S.statBox}><div style={{fontSize:9, color:'#444', marginBottom:3}}>ENTRADA</div><div style={{fontSize:14, fontWeight:700}}>$414</div></div>
              <div style={S.statBox}><div style={{fontSize:9, color:'#444', marginBottom:3}}>STOP LOSS</div><div style={{fontSize:14, fontWeight:700, color:'#D85A30'}}>$407</div></div>
              <div style={S.statBox}><div style={{fontSize:9, color:'#444', marginBottom:3}}>TARGET</div><div style={{fontSize:14, fontWeight:700, color:'#1D9E75'}}>$430</div></div>
            </div>

            <div style={S.rsiGrid}>
              {[['RSI 6','61.8'],['RSI 12','58.2'],['RSI 24','54.1'],['MA5','412'],['MA10','409']].map(([l,v])=>
                <div key={l} style={S.rsiBox}>
                  <div style={{fontSize:8, color:'#444', marginBottom:2}}>{l}</div>
                  <div style={{fontSize:11, fontWeight:700}}>{v}</div>
                </div>
              )}
            </div>
          </div>
        )}

        {/* TAB LUNA */}
        {tab==='luna' && (
          <div style={{display:'flex', flexDirection:'column', height:'calc(100vh - 200px)'}}>
            {/* Luna header */}
            <div style={{display:'flex', alignItems:'center', gap:10, marginBottom:10, padding:'10px 12px', background:'#111', borderRadius:10, border:'1px solid #1a1a1a'}}>
              <div style={{width:36, height:36, borderRadius:'50%', background:'#1a1040', display:'flex', alignItems:'center', justifyContent:'center', fontSize:18, flexShrink:0}}>🌙</div>
              <div style={{flex:1}}>
                <div style={{fontSize:13, fontWeight:700}}>Luna</div>
                <div style={{fontSize:10, color:'#1D9E75'}}><span style={S.dot}></span>Trade Coach · en linea</div>
              </div>
              <div style={S.modeRow}>
                <button onClick={()=>setMode('long')} style={S.modeBtn(mode==='long','#1D9E75')}>▲ LONG</button>
                <button onClick={()=>setMode('short')} style={S.modeBtn(mode==='short','#e05252')}>▼ SHORT</button>
              </div>
            </div>

            {mode==='short' && (
              <div style={{background:'rgba(224,82,82,0.07)', border:'1px solid rgba(224,82,82,0.2)', borderRadius:8, padding:'6px 12px', marginBottom:8, fontSize:10, color:'#e05252'}}>
                ⚠️ MODO SHORT — Luna verifica las 3 condiciones antes de aprobar.
              </div>
            )}

            {/* Chat */}
            <div ref={chatRef} style={{flex:1, overflowY:'auto', display:'flex', flexDirection:'column', gap:8, marginBottom:8}}>

                  </div>
                ))}
              </div>
            )}

            {/* Input */}
            <div style={S.inputArea}>
              <input ref={fileRef} type="file" accept="image/*" multiple style={{display:'none'}} onChange={handleImages}/>
              <div style={S.inputRow}>
                <button onClick={()=>fileRef.current.click()} style={S.iconBtn('#c9a84c')}>📷</button>
                <input
                  value={inp}
                  onChange={e=>setInp(e.target.value)}
                  onKeyDown={e=>e.key==='Enter'&&send()}
                  placeholder="Pregunta a Luna o sube charts..."
                  style={S.chatInput}
                />
                <button onClick={send} disabled={loading} style={{...S.sendBtn(mode), opacity:loading?0.4:1}}>
                  {loading?'...':'►'}
                </button>
              </div>
            </div>
          </div>
        )}

        {/* TAB ADMIN */}
        {tab==='admin' && (
          <div>
            <div style={{fontSize:13, fontWeight:700, color:'#c9a84c', marginBottom:12, letterSpacing:1}}>⚙ ADMINISTRADOR</div>
            {!adminAuth ? (
              <div style={S.adminCard}>
                <div style={{fontSize:11, color:'#888', marginBottom:10}}>Contraseña de acceso:</div>
                <input
                  type="password"
                  placeholder="Contraseña..."
                  value={adminPass}
                  onChange={e=>setAdminPass(e.target.value)}
                  onKeyDown={e=>e.key==='Enter'&&loginAdmin()}
                  style={{...S.chatInput, width:'100%', marginBottom:10, boxSizing:'border-box'}}
                />
                <button onClick={loginAdmin} style={{width:'100%', padding:'10px', borderRadius:8, border:'1px solid #c9a84c', background:'transparent', color:'#c9a84c', fontSize:13, fontWeight:700, cursor:'pointer'}}>ENTRAR</button>
                {adminMsg&&<div style={{fontSize:11, color:'#e05252', marginTop:8}}>{adminMsg}</div>}
              </div>
            ) : (
              <div>
                <div style={{...S.card, marginBottom:10}}>
                  <div style={{fontSize:10, color:'#888', letterSpacing:1, marginBottom:8}}>ACTIVOS ACTIVOS</div>
                  <div style={{display:'flex', flexDirection:'column', gap:6}}>
                    {assets.map(a=>{
                      const p=prices[a.n]
                      return(
                        <div key={a.n} style={{display:'flex', alignItems:'center', justifyContent:'space-between', background:'#0d0d0d', border:'1px solid #1a1a1a', borderRadius:8, padding:'8px 12px'}}>
                          <div>
                            <div style={{fontSize:12, fontWeight:700}}>{a.label} <span style={{fontSize:10, color:'#444'}}>({a.n})</span></div>
                            <div style={{fontSize:11, color:p?.up?'#1D9E75':'#D85A30'}}>{p?'$'+p.price:'cargando...'}</div>
                          </div>
                          <button onClick={()=>removeAsset(a.n)} style={{fontSize:11, background:'transparent', border:'1px solid #e05252', borderRadius:6, color:'#e05252', cursor:'pointer', padding:'4px 10px'}}>Quitar</button>
                        </div>
                      )
                    })}
                  </div>
                </div>

                <div style={S.adminCard}>
                  <div style={{fontSize:10, color:'#888', letterSpacing:1, marginBottom:10}}>AGREGAR ACTIVO</div>
                  <input
                    placeholder="Simbolo Bitget (ej: ADAUSDT)"
                    value={newSymbol}
                    onChange={e=>setNewSymbol(e.target.value)}
                    style={{...S.chatInput, width:'100%', marginBottom:8, boxSizing:'border-box'}}
                  />
                  <input
                    placeholder="Nombre corto (ej: ADA)"
                    value={newLabel}
                    onChange={e=>setNewLabel(e.target.value)}
                    style={{...S.chatInput, width:'100%', marginBottom:10, boxSizing:'border-box'}}
                  />
                  <button onClick={addAsset} style={{width:'100%', padding:'10px', borderRadius:8, border:'1px solid #1D9E75', background:'transparent', color:'#1D9E75', fontSize:13, fontWeight:700, cursor:'pointer'}}>+ AGREGAR ACTIVO</button>
                  {adminMsg&&<div style={{fontSize:11, color:adminMsg.includes('✅')?'#1D9E75':'#e05252', marginTop:8}}>{adminMsg}</div>}
                  <div style={{fontSize:9, color:'#333', marginTop:8}}>El simbolo debe existir en Bitget Spot</div>
                </div>
              </div>
            )}
          </div>
        )}
      </div>

      {/* BOTTOM NAV */}
      <div style={S.bottomNav}>
        {[
          {id:'mercado', icon:'📊', label:'MERCADO'},
          {id:'luna',    icon:'🌙', label:'LUNA'},
          {id:'admin',   icon:'⚙️', label:'ADMIN'},
        ].map(n=>(
          <button key={n.id} onClick={()=>setTab(n.id)} style={S.navBtn(tab===n.id)}>
            <span style={{fontSize:18}}>{n.icon}</span>
            <span>{n.label}</span>
          </button>
        ))}
      </div>

    </div>
  )
}
