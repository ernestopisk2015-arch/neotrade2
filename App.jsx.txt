                        {m.imgs&&m.imgs.length>0&&(
                          <div style={{display:'flex', gap:4, flexWrap:'wrap', marginBottom:4, justifyContent:m.f==='l'?'flex-start':'flex-end'}}>
                            {m.imgs.map((img,idx)=><img key={idx} src={img} style={{height:52, width:68, borderRadius:6, objectFit:'cover', border:'1px solid #222'}} alt="chart"/>)}
                          </div>
                        )}
                        <div className={m.f==='l'?'bubble-luna':'bubble-user'}>{m.t}</div>
                        <div className="msg-time">{m.f==='l'?'LUNA':'TÚ'}</div>
                      </div>
                    ))}
                    {loading&&<div className="bubble-luna" style={{color:'#444'}}>Luna analizando...</div>}
                  </div>
                  {images.length>0&&(
                    <div style={{display:'flex', gap:6, flexWrap:'wrap', marginBottom:8}}>
                      {images.map((img,i)=>(
                        <div key={i} style={{position:'relative'}}>
                          <img src={img} style={{height:48, width:64, objectFit:'cover', borderRadius:6, border:'1px solid #c9a84c'}} alt="preview"/>
                          <button onClick={()=>removeImage(i)} style={{position:'absolute',top:-4,right:-4,width:16,height:16,borderRadius:'50%',background:'#D85A30',border:'none',color:'white',fontSize:9,cursor:'pointer'}}>✕</button>
                        </div>
                      ))}
                    </div>
                  )}
                  <div className="input-area" style={{padding:0}}>
                    <input ref={fileRef} type="file" accept="image/*" multiple style={{display:'none'}} onChange={handleImages}/>
                    <div className="input-row">
                      <button onClick={()=>fileRef.current.click()} className="icon-btn">📷</button>
                      <input value={inp} onChange={e=>setInp(e.target.value)} onKeyDown={e=>e.key==='Enter'&&send()} placeholder="Pregunta a Luna o sube charts..." className="chat-input"/>
                      <button onClick={send} disabled={loading} style={{fontSize:12, padding:'9px 16px', borderRadius:6, border:`1px solid ${mode==='short'?'#e05252':'#c9a84c'}`, background:'transparent', color:mode==='short'?'#e05252':'#c9a84c', cursor:'pointer', fontWeight:700, opacity:loading?0.4:1, fontFamily:'Courier New'}}>
                        {loading?'...':'►'}
                      </button>
                    </div>
                  </div>
                </div>
              )}

              {/* TAB ADMIN */}
              {tab==='admin' && (
                <div style={{maxWidth:500}}>
                  <div style={{fontSize:13, fontWeight:700, color:'#c9a84c', letterSpacing:2, marginBottom:14}}>⚙ ADMINISTRADOR</div>
                  {!adminAuth ? (
                    <div className="card">
                      <div className="section-title">CONTRASEÑA DE ACCESO</div>
                      <input type="password" placeholder="Contraseña..." value={adminPass} onChange={e=>setAdminPass(e.target.value)} onKeyDown={e=>e.key==='Enter'&&loginAdmin()} className="admin-input"/>
                      <button onClick={loginAdmin} className="btn-gold">ENTRAR</button>
                      {adminMsg&&<div style={{fontSize:11, color:'#e05252', marginTop:8}}>{adminMsg}</div>}
                    </div>
                  ) : (
                    <div>
                      <div className="card">
                        <div className="section-title">ACTIVOS ACTIVOS</div>
                        {assets.map(a=>{
                          const p=prices[a.n]
                          return(
                            <div key={a.n} className="asset-row">
                              <div>
                                <div style={{fontSize:13, fontWeight:700}}>{a.label} <span style={{fontSize:10, color:'#444'}}>({a.n})</span></div>
                                <div style={{fontSize:11, color:p?.up?'#1D9E75':'#D85A30'}}>{p?'$'+p.price:'cargando...'}</div>
                              </div>
                              <button onClick={()=>removeAsset(a.n)} className="remove-btn">Quitar</button>
                            </div>
                          )
                        })}
                      </div>
                      <div className="card" style={{border:'1px solid #c9a84c'}}>
                        <div className="section-title">AGREGAR ACTIVO</div>
                        <input placeholder="Simbolo Bitget (ej: ADAUSDT, RKLBUSDT)" value={newSymbol} onChange={e=>setNewSymbol(e.target.value)} className="admin-input"/>
                        <input placeholder="Nombre corto (ej: ADA, RKLB)" value={newLabel} onChange={e=>setNewLabel(e.target.value)} className="admin-input"/>
                        <button onClick={addAsset} className="btn-green">+ AGREGAR ACTIVO</button>
                        {adminMsg&&<div style={{fontSize:11, color:adminMsg.includes('✅')?'#1D9E75':'#e05252', marginTop:8}}>{adminMsg}</div>}
                        <div style={{fontSize:9, color:'#333', marginTop:8}}>Crypto: BTCUSDT, ETHUSDT / Acciones: RKLBUSDT, RGTIUSDT</div>
                      </div>
                    </div>
                  )}
                </div>
              )}
            </div>
          </div>

          {/* SIDEBAR LUNA — SOLO DESKTOP */}
          <div className="sidebar">
            <div className="luna-header">
              <div className="luna-avatar">🌙</div>
              <div style={{flex:1}}>
                <div style={{fontSize:13, fontWeight:700}}>Luna</div>
                <div style={{fontSize:10, color:'#1D9E75'}}><span className="live-dot"></span>Trade Coach · en linea</div>
              </div>
              <div className="mode-row">
                <button onClick={()=>setMode('long')} className="mode-btn" style={{border:`1px solid ${mode==='long'?'#1D9E75':'#222'}`, background:mode==='long'?'rgba(29,158,117,0.1)':'transparent', color:mode==='long'?'#1D9E75':'#444'}}>▲ LONG</button>
                <button onClick={()=>setMode('short')} className="mode-btn" style={{border:`1px solid ${mode==='short'?'#e05252':'#222'}`, background:mode==='short'?'rgba(224,82,82,0.1)':'transparent', color:mode==='short'?'#e05252':'#444'}}>▼ SHORT</button>
              </div>
            </div>

            {mode==='short'&&<div className="short-warning" style={{margin:'8px 14px 0'}}>⚠️ MODO SHORT — 3 condiciones requeridas.</div>}

            <div ref={chatRef} className="chat-area">
              {msgs.map((m,i)=>(
                <div key={i} style={{display:'flex', flexDirection:'column', alignItems:m.f==='l'?'flex-start':'flex-end'}}>
                  {m.imgs&&m.imgs.length>0&&(
                    <div style={{display:'flex', gap:4, flexWrap:'wrap', marginBottom:4, justifyContent:m.f==='l'?'flex-start':'flex-end'}}>
                      {m.imgs.map((img,idx)=><img key={idx} src={img} style={{height:52, width:68, borderRadius:6, objectFit:'cover', border:'1px solid #222'}} alt="chart"/>)}
                    </div>
                  )}
                  <div className={m.f==='l'?'bubble-luna':'bubble-user'}>{m.t}</div>
                  <div className="msg-time">{m.f==='l'?'LUNA':'TÚ'}</div>
                </div>
              ))}
              {loading&&<div className="bubble-luna" style={{color:'#444'}}>Luna analizando...</div>}
            </div>

            {images.length>0&&(
              <div style={{display:'flex', gap:6, flexWrap:'wrap', padding:'8px 14px', borderTop:'1px solid #1a1a1a'}}>
                {images.map((img,i)=>(
                  <div key={i} style={{position:'relative'}}>
                    <img src={img} style={{height:48, width:64, objectFit:'cover', borderRadius:6, border:'1px solid #c9a84c'}} alt="preview"/>
                    <button onClick={()=>removeImage(i)} style={{position:'absolute',top:-4,right:-4,width:16,height:16,borderRadius:'50%',background:'#D85A30',border:'none',color:'white',fontSize:9,cursor:'pointer'}}>✕</button>
                  </div>
                ))}
              </div>
            )}

            <div className="input-area">
              <input ref={fileRef} type="file" accept="image/*" multiple style={{display:'none'}} onChange={handleImages}/>
              <div className="input-row">
                <button onClick={()=>fileRef.current.click()} className="icon-btn">📷</button>
                <input value={inp} onChange={e=>setInp(e.target.value)} onKeyDown={e=>e.key==='Enter'&&send()} placeholder="Pregunta a Luna..." className="chat-input"/>
                <button onClick={send} disabled={loading} style={{fontSize:12, padding:'9px 16px', borderRadius:6, border:`1px solid ${mode==='short'?'#e05252':'#c9a84c'}`, background:'transparent', color:mode==='short'?'#e05252':'#c9a84c', cursor:'pointer', fontWeight:700, opacity:loading?0.4:1, fontFamily:'Courier New'}}>
                  {loading?'...':'►'}
                </button>
              </div>
            </div>
          </div>
        </div>

        {/* BOTTOM NAV — SOLO MOBILE */}
        <div className="bottom-nav">
          {[{id:'mercado',icon:'📊',label:'MERCADO'},{id:'luna',icon:'🌙',label:'LUNA'},{id:'admin',icon:'⚙️',label:'ADMIN'}].map(n=>(
            <button key={n.id} onClick={()=>setTab(n.id)} className={`nav-btn${tab===n.id?' active':''}`}>
              <span style={{fontSize:18}}>{n.icon}</span>
              <span>{n.label}</span>
            </button>
          ))}
        </div>
      </div>
    </>
  )
}
